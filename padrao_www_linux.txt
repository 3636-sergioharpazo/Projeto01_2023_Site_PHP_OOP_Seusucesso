# Configuração para o domínio chatceecegril.antoniooliveira.shop (HTTPS na porta 8443)
server {
    listen 8443 ssl;
    listen [::]:8443 ssl;

    server_name chatceecegril.antoniooliveira.shop;

    # Configuração do SSL
    ssl_certificate /etc/ssl/certs/nginx-selfsigned.crt;
    ssl_certificate_key /etc/ssl/private/nginx-selfsigned.key;

    # Raiz do site
    root /var/www/html;
    index index.html index.htm;

    # Proxy para o serviço na porta 3002
    location / {
        proxy_pass http://127.0.0.1:3002;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }
}

# Redirecionar HTTP para HTTPS para chatceecegril.antoniooliveira.shop (porta 8080 para 8443)
server {
    listen 8080;
    listen [::]:8080;

    server_name chatceecegril.antoniooliveira.shop;

    return 301 https://$host:8443$request_uri;  # Redireciona HTTP para HTTPS
}
