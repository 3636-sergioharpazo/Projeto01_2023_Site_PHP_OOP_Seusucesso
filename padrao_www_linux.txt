server {
    listen 443 ssl default_server;
    listen [::]:443 ssl default_server;

    server_name 92.112.179.191;

    # Configuração do SSL
    ssl_certificate /etc/ssl/certs/nginx-selfsigned.crt;
    ssl_certificate_key /etc/ssl/private/nginx-selfsigned.key;

    # Raiz do site
    root /var/www/html;
    index index.html index.htm;

    # Servir diretamente o arquivo qrcode.html
    location = /qrcode.html {
        root /var/www/html;  # Define o diretório raiz
        try_files $uri =404;  # Verifica se o arquivo existe e serve diretamente
    }

    # Servir diretamente o arquivo qrcode.png
    location = /qrcode.png {
        root /var/www/html;
        try_files $uri =404;
    }

    # Proxy para o serviço na porta 3002
    location / {
        proxy_pass http://127.0.0.1:3002;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }
}
